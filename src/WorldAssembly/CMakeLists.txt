set(WORLD_ASSEMBLY_INCLUDE_PATH ${ROOT_INCLUDE_PATH}/hgl/world/assembly)

file(GLOB_RECURSE WORLD_ASSEMBLY_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
file(GLOB_RECURSE WORLD_ASSEMBLY_PRIVATE_HEADERS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/*.inl")
file(GLOB_RECURSE WORLD_ASSEMBLY_PUBLIC_HEADERS CONFIGURE_DEPENDS
    "${WORLD_ASSEMBLY_INCLUDE_PATH}/*.h")

source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}"
    PREFIX "WorldAssembly\\Implementation"
    FILES ${WORLD_ASSEMBLY_SOURCES} ${WORLD_ASSEMBLY_PRIVATE_HEADERS})
source_group(TREE "${WORLD_ASSEMBLY_INCLUDE_PATH}"
    PREFIX "WorldAssembly\\Public"
    FILES ${WORLD_ASSEMBLY_PUBLIC_HEADERS})

add_cm_library(ULRE.WorldAssembly "ULRE" ${WORLD_ASSEMBLY_INCLUDE_PATH}
    ${WORLD_ASSEMBLY_SOURCES}
    ${WORLD_ASSEMBLY_PRIVATE_HEADERS}
    ${WORLD_ASSEMBLY_PUBLIC_HEADERS})

target_include_directories(ULRE.WorldAssembly PUBLIC ${ROOT_INCLUDE_PATH})
# 公开头（OffscreenWorld/LevelLoader/Gizmo）直接暴露 ECS、SceneGraph、Vulkan
# 类型 ⇒ 三条边都必须是 PUBLIC：消费方需要同样的头与目标，而不是靠根聚合。
target_link_libraries(ULRE.WorldAssembly
    PUBLIC
        ULRE.ECS
        ULRE.SceneGraph
        ULRE.Vulkan)
